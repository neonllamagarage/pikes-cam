function openItemStudioDirect(item, forcedBin) { selectedAuditItem = item; isUnlistedActive = false; auditShots = [null, null, null]; setAuditStep(0); document.getElementById('slot-tag-0').innerText = "READY"; document.getElementById('slot-tag-1').innerText = "WAIT"; document.getElementById('slot-tag-2').innerText = "WAIT"; document.getElementById('studioItemTitle').innerText = item.title; document.getElementById('studioItemSku').innerText = "SKU: " + item.sku; document.getElementById('studioQtyInput').value = item.qty || 1; // Clean the bin input to single clean box (e.g. HC16) var cleanBin = forcedBin || 'HC16'; if (/^\d+$/.test(cleanBin)) cleanBin = 'HC' + cleanBin.padStart(2, '0'); document.getElementById('studioBinInput').value = cleanBin.toUpperCase(); // Resolve size var resolvedSize = (item.size && item.size.trim().length > 0) ? item.size.trim() : parseItemSize(item.title); var sel = document.getElementById('studioSizeSelect'); var customInp = document.getElementById('studioCustomSizeInput'); customInp.classList.add('hidden'); customInp.value = ''; var matchedOpt = Array.from(sel.options).find(function(opt) { return opt.value.toLowerCase() === resolvedSize.toLowerCase(); }); if (matchedOpt) { sel.value = matchedOpt.value; } else { sel.value = 'CUSTOM'; customInp.classList.remove('hidden'); customInp.value = resolvedSize; } document.getElementById('auditRefThumb').src = item.imageUrl || 'https://via.placeholder.com/100'; document.getElementById('auditOverlayBox').classList.remove('hidden'); document.getElementById('auditTargetBadge').classList.remove('hidden'); var cam = document.getElementById('viewfinderWrapper'); if (cam) cam.classList.remove('hidden'); document.getElementById('auditStageBinSelect').classList.add('hidden'); document.getElementById('auditCardDeck').classList.add('hidden'); document.getElementById('unlistedStudioControls').classList.add('hidden'); document.getElementById('auditStudioControls').classList.remove('hidden'); }
